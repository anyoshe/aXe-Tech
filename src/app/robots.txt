import { MetadataRoute } from 'next';

export default function robots(): MetadataRoute.Robots {
  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: [
        '/api/',
        '/admin/',
        '/dashboard/',
        '/_next/',
        '/private/',
        '/studio/',
        '/_next/static/',
        '/_next/image/',
      ],
    },
    sitemap: 'https://getaxekenya.com/sitemap.xml',
    host: 'https://getaxekenya.com',
  };
}
